package model;

/**
 * Abstract base class representing a person in the SmartStay system.
 * Encapsulates common identity and contact details shared across system roles.
 */
public abstract class E195569Person {
    private final String fullName;
    private final String contactNumber;
    private final String email;

    /**
     * Constructs a validated Person instance.
     *
     * @param fullName      Full legal name of the person.
     * @param contactNumber Contact telephone number.
     * @param email         Contact email address.
     * @throws IllegalArgumentException if any argument is null or blank.
     */
    public E195569Person(String fullName, String contactNumber, String email) {
        if (fullName == null || fullName.isBlank()) {
            throw new IllegalArgumentException("Full name cannot be null or blank.");
        }
        if (contactNumber == null || contactNumber.isBlank()) {
            throw new IllegalArgumentException("Contact number cannot be null or blank.");
        }
        if (email == null || email.isBlank()) {
            throw new IllegalArgumentException("Email cannot be null or blank.");
        }

        this.fullName = fullName.trim();
        this.contactNumber = contactNumber.trim();
        this.email = email.trim();
    }

    public String getFullName() {
        return fullName;
    }

    public String getContactNumber() {
        return contactNumber;
    }

    public String getEmail() {
        return email;
    }

    /**
     * Abstract method to obtain formatted details specific to the concrete role.
     *
     * @return Formatted detail string.
     */
    public abstract String getDetails();
}

package model;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.temporal.ChronoUnit;
import java.util.Objects;

/**
 * Represents a reservation record in the SmartStay system.
 * Connects the guest, assigned room, stay duration, advance payment, and reservation lifecycle state.
 */
public class E195569Booking {

    /** Standard refundable booking advance payment required at reservation (LKR 10,000). */
    private static final double STANDARD_ADVANCE = 10000.0;

    /** Window in hours prior to check-in midnight when cancellation/modification is blocked. */
    private static final int CANCELLATION_DEADLINE_HOURS = 24;

    /** Minimum required number of guests for any valid booking. */
    private static final int MIN_GUESTS = 1;

    private final String bookingId;
    private final E195569Guest guest;
    private final E195569Room room;
    private final LocalDate checkInDate;
    private final LocalDate checkOutDate;
    private final int numberOfGuests;
    private final double advancePaid;
    private E195569BookingStatus bookingStatus;

    /**
     * Constructs a validated booking instance with standard advance payment of LKR 10,000.
     *
     * @param bookingId      Unique booking identifier (e.g., "BK-1001").
     * @param guest          Registered guest holding the reservation.
     * @param room           Allocated room for the stay.
     * @param checkInDate    Start date of the reservation.
     * @param checkOutDate   End date of the reservation.
     * @param numberOfGuests Number of occupants staying in the room.
     * @throws IllegalArgumentException if arguments are null/blank, guests < 1, or check-out is not after check-in.
     */
    public E195569Booking(String bookingId, E195569Guest guest, E195569Room room,
                          LocalDate checkInDate, LocalDate checkOutDate, int numberOfGuests) {
        this(bookingId, guest, room, checkInDate, checkOutDate, numberOfGuests, E195569BookingStatus.CONFIRMED);
    }

    /**
     * Overloaded constructor supporting initial status specification.
     */
    public E195569Booking(String bookingId, E195569Guest guest, E195569Room room,
                          LocalDate checkInDate, LocalDate checkOutDate, int numberOfGuests,
                          E195569BookingStatus bookingStatus) {
        if (bookingId == null || bookingId.isBlank()) {
            throw new IllegalArgumentException("Booking ID cannot be null or blank.");
        }
        this.guest = Objects.requireNonNull(guest, "Guest cannot be null.");
        this.room = Objects.requireNonNull(room, "Room cannot be null.");
        this.checkInDate = Objects.requireNonNull(checkInDate, "Check-in date cannot be null.");
        this.checkOutDate = Objects.requireNonNull(checkOutDate, "Check-out date cannot be null.");

        if (!checkOutDate.isAfter(checkInDate)) {
            throw new IllegalArgumentException("Check-out date (" + checkOutDate + ") must be after check-in date (" + checkInDate + ").");
        }
        if (numberOfGuests < MIN_GUESTS) {
            throw new IllegalArgumentException("Number of guests must be at least " + MIN_GUESTS + ", but received " + numberOfGuests + ".");
        }

        this.bookingId = bookingId.trim();
        this.numberOfGuests = numberOfGuests;
        this.advancePaid = STANDARD_ADVANCE;
        this.bookingStatus = Objects.requireNonNull(bookingStatus, "Booking status cannot be null.");
    }

    public String getBookingId() {
        return bookingId;
    }

    public E195569Guest getGuest() {
        return guest;
    }

    public E195569Room getRoom() {
        return room;
    }

    public LocalDate getCheckInDate() {
        return checkInDate;
    }

    public LocalDate getCheckOutDate() {
        return checkOutDate;
    }

    public int getNumberOfGuests() {
        return numberOfGuests;
    }

    public double getAdvancePaid() {
        return advancePaid;
    }

    public E195569BookingStatus getBookingStatus() {
        return bookingStatus;
    }

    /**
     * Calculates the total duration of the stay in nights.
     *
     * @return Number of nights between check-in and check-out dates.
     */
    public long getNights() {
        return ChronoUnit.DAYS.between(checkInDate, checkOutDate);
    }

    /**
     * Determines whether the reservation can be modified or cancelled based on business rules.
     * Allowed only if status is CONFIRMED and more than 24 hours remain before check-in at 00:00.
     *
     * @param now Current date and time reference.
     * @return true if cancellation or modification is permitted; false otherwise.
     */
    public boolean canModifyOrCancel(LocalDateTime now) {
        if (bookingStatus != E195569BookingStatus.CONFIRMED) {
            return false;
        }
        LocalDateTime checkInMidnight = checkInDate.atStartOfDay();
        return now.isBefore(checkInMidnight.minusHours(CANCELLATION_DEADLINE_HOURS));
    }

    /**
     * Checks if a candidate date range conflicts with this booking's date interval.
     *
     * @param in  Candidate check-in date.
     * @param out Candidate check-out date.
     * @return true if the date ranges overlap, false otherwise.
     */
    // Algorithm: Interval Overlap Detection - two half-open intervals [in, out) and [checkInDate, checkOutDate) overlap if in < checkOutDate && out > checkInDate
    public boolean overlaps(LocalDate in, LocalDate out) {
        if (in == null || out == null || !out.isAfter(in)) {
            throw new IllegalArgumentException("Invalid date range supplied for overlap check.");
        }
        return in.isBefore(checkOutDate) && out.isAfter(checkInDate);
    }

    /**
     * Transitions the reservation status to CANCELLED.
     */
    public void cancel() {
        this.bookingStatus = E195569BookingStatus.CANCELLED;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        E195569Booking that = (E195569Booking) o;
        return Objects.equals(bookingId, that.bookingId);
    }

    @Override
    public int hashCode() {
        return Objects.hash(bookingId);
    }

    @Override
    public String toString() {
        return String.format("| %-9s | %-7s | %-16s | %-10s | %-10s | %2d night(s) | %2d guest(s) | LKR %,9.2f | %-9s |",
                bookingId, room.getRoomId(), guest.getFullName(), checkInDate, checkOutDate,
                getNights(), numberOfGuests, advancePaid, bookingStatus.getDisplayName());
    }
}

    private final Map<String, E195569Room> rooms = new HashMap<>();

    /**
     * Adds a new room to the repository, rejecting duplicate IDs.
     *
     * @param room The room to add.

package service;

import exception.E195569BookingException;

/**
 * Service managing reservation lifecycles, availability, searching, and invoice generation.
 */
public class E195569BookingService {

    /** Minimum lead time in days required between booking creation and check-in date. */
    private static final int MIN_LEAD_TIME_DAYS = 2;

    /** Minimum required number of guests for any valid booking. */
    private static final int MIN_GUESTS = 1;

    // Data Structure: ArrayList holding the in-memory sequence of all booking records
    private final List<E195569Booking> bookings = new ArrayList<>();

    private final E195569RoomService roomService;
    private final E195569GuestService guestService;
    private final E195569InvoiceCalculator invoiceCalculator;
    private int bookingCounter = 1;

    public E195569BookingService(E195569RoomService roomService, E195569GuestService guestService,
                                 E195569InvoiceCalculator invoiceCalculator) {
        this.roomService = Objects.requireNonNull(roomService, "Room service cannot be null.");
        this.guestService = Objects.requireNonNull(guestService, "Guest service cannot be null.");
        this.invoiceCalculator = Objects.requireNonNull(invoiceCalculator, "Invoice calculator cannot be null.");
    }

    /**
     * Creates a new booking by locating the first available room of the requested type without overlapping reservations.
     *
     * @param nicOrPassport  Guest identifier.
     * @param roomType       Requested room category.
     * @param numberOfGuests Number of occupants.
     * @param checkIn        Reservation start date.
     * @param checkOut       Reservation end date.
     * @return Newly created and stored E195569Booking instance.
     * @throws E195569BookingException if any business rule is violated or no room is free.
     */
    // Algorithm: First-fit linear search combined with interval overlap detection
    public E195569Booking createBooking(String nicOrPassport, E195569RoomType roomType, int numberOfGuests,
                                        LocalDate checkIn, LocalDate checkOut) throws E195569BookingException {
        E195569Guest guest = guestService.findByNic(nicOrPassport);
        if (numberOfGuests < MIN_GUESTS) {
            throw new E195569BookingException("Number of guests must be at least " + MIN_GUESTS + ", but received " + numberOfGuests + ".");
        }
        validateDateRange(checkIn, checkOut);
        LocalDate earliestAllowed = LocalDate.now().plusDays(MIN_LEAD_TIME_DAYS);
        if (checkIn.isBefore(earliestAllowed)) {
            throw new E195569BookingException("Check-in must be at least 2 days after today (" + earliestAllowed + " onwards).");
        }
        if (roomType == null) {
            throw new E195569BookingException("Room type cannot be null.");
        }

        E195569Room assignedRoom = null;
        for (E195569Room r : roomService.searchAvailable(roomType)) {
            if (!hasConflictingBooking(r, checkIn, checkOut)) {
                assignedRoom = r;
                break;
            }
        }

        if (assignedRoom == null) {
            throw new E195569BookingException("No available room of type " + roomType.getDisplayName() + " for dates " + checkIn + " to " + checkOut + ".");
        }

        String bookingId = String.format("BK-%04d", bookingCounter++);
        E195569Booking booking = new E195569Booking(bookingId, guest, assignedRoom, checkIn, checkOut, numberOfGuests);
        assignedRoom.setStatus(E195569RoomStatus.RESERVED);
        bookings.add(booking);
        return booking;
    }

    /**
     * Searches for all rooms of a given type with status AVAILABLE and no overlapping CONFIRMED bookings.
     *
     * @param type     Room category.
     * @param checkIn  Desired check-in date.
     * @param checkOut Desired check-out date.
     * @return List of matching available rooms.
     * @throws E195569BookingException if check-out date is not after check-in date or dates are null.
     */
    // Algorithm: Linear filtering with interval overlap conflict detection
    public List<E195569Room> searchAvailableRooms(E195569RoomType type, LocalDate checkIn, LocalDate checkOut)
            throws E195569BookingException {
        validateDateRange(checkIn, checkOut);
        if (type == null) {
            throw new E195569BookingException("Room type cannot be null.");
        }

        List<E195569Room> available = new ArrayList<>();
        for (E195569Room r : roomService.searchAvailable(type)) {
            if (!hasConflictingBooking(r, checkIn, checkOut)) {
                available.add(r);
            }
        }
        return Collections.unmodifiableList(available);
    }

    /**
     * Cancels an existing booking if outside the 24-hour check-in window.
     *
     * @param bookingId Booking identifier.
     * @return Cancelled booking instance.
     * @throws E195569BookingException if not found or within 24 hours of check-in at 00:00.
     */
    // Algorithm: Linear search for target booking and state transition
    public E195569Booking cancelBooking(String bookingId) throws E195569BookingException {
        E195569Booking booking = searchByBookingId(bookingId);
        if (!booking.canModifyOrCancel(LocalDateTime.now())) {
            throw new E195569BookingException("Cancellation rejected: Cannot cancel within 24 hours of check-in at 00:00 (Rule: >24h required).");
        }

        booking.cancel();
        booking.getRoom().setStatus(E195569RoomStatus.AVAILABLE);
        return booking;
    }

    /**
     * Searches for a booking by its exact ID using linear search.
     *
     * @param id Booking identifier.
     * @return Matching booking.
     * @throws E195569BookingException if not found.
     */
    // Algorithm: Linear Search for exact-key match
    public E195569Booking searchByBookingId(String id) throws E195569BookingException {
        if (id == null || id.isBlank()) {
            throw new E195569BookingException("Booking ID cannot be blank.");
        }
        for (E195569Booking b : bookings) {
            if (b.getBookingId().equalsIgnoreCase(id.trim())) {
                return b;
            }
        }
        throw new E195569BookingException("Booking with ID '" + id + "' was not found.");
    }

    /**
     * Searches bookings matching guest name partially (case-insensitive).
     *
     * @param name Guest name or substring.
     * @return List of matching bookings.
     */
    // Algorithm: Linear Search with case-insensitive partial substring match
    public List<E195569Booking> searchByGuestName(String name) {
        if (name == null || name.isBlank()) {
            return Collections.emptyList();
        }
        String query = name.trim().toLowerCase();
        List<E195569Booking> matching = new ArrayList<>();
        for (E195569Booking b : bookings) {
            if (b.getGuest().getFullName().toLowerCase().contains(query)) {
                matching.add(b);
            }
        }
        return Collections.unmodifiableList(matching);
    }

    /**
     * Returns a copy of all bookings sorted chronologically by check-in date.
     *
     * @return Sorted copy of bookings list.
     */
    // Algorithm: Comparator-based TimSort on checkInDate
    public List<E195569Booking> viewByCheckInDate() {
        List<E195569Booking> sorted = new ArrayList<>(bookings);
        sorted.sort(Comparator.comparing(E195569Booking::getCheckInDate));
        return Collections.unmodifiableList(sorted);
    }

    /**
     * Generates an itemized invoice for the requested booking.
     *
     * @param bookingId Booking identifier.
     * @return E195569Invoice instance computed by E195569InvoiceCalculator.
     * @throws E195569BookingException if booking is not found.
     */
    // Algorithm: Linear search for booking ID followed by direct formula calculation
    public E195569Invoice generateInvoice(String bookingId) throws E195569BookingException {
        E195569Booking booking = searchByBookingId(bookingId);
        return invoiceCalculator.calculate(booking);
    }

    /**
     * Helper checking for date collisions on a room against active confirmed bookings.
     */
    // Algorithm: Interval Overlap Detection (in < otherOut && out > otherIn)
    private boolean hasConflictingBooking(E195569Room room, LocalDate in, LocalDate out) {
        for (E195569Booking b : bookings) {
            if (b.getRoom().equals(room) && b.getBookingStatus() == E195569BookingStatus.CONFIRMED) {
                if (b.overlaps(in, out)) {
                    return true;
                }
            }
        }
        return false;
    }

    /**
     * Validates that dates are non-null and check-out is strictly after check-in.
     */
    private void validateDateRange(LocalDate checkIn, LocalDate checkOut) throws E195569BookingException {
        if (checkIn == null || checkOut == null) {
            throw new E195569BookingException("Check-in and check-out dates cannot be null.");
        }
        if (!checkOut.isAfter(checkIn)) {
            throw new E195569BookingException("Check-out date (" + checkOut + ") must be after check-in date (" + checkIn + ").");
        }
    }
}

package service;

import java.util.Objects;
import model.E195569Booking;
import model.E195569Invoice;
import model.E195569Room;
import model.E195569RoomType;

/**
 * Service calculating itemized invoice totals for reservations according to coursework rules.
 */
public class E195569InvoiceCalculator {

    /** Standard refundable booking advance payment required at reservation. */
    public static final double ADVANCE = 10000.0;

    /** Discount percentage (12%) applied to base room charge for stays >= 5 nights. */
    public static final double DISCOUNT_RATE = 0.12;

    /** Minimum stay duration in nights required to qualify for long stay discount. */
    public static final int DISCOUNT_THRESHOLD_NIGHTS = 5;

    /**
     * Calculates the itemized invoice for a given booking.
     * Enforces the exact formula:
     * - base = dailyRate x nights
     * - extra = max(0, guests - capacity) x extraGuestCharge x nights
     * - discount = nights >= 5 ? DISCOUNT_RATE x base : 0
     * - taxable = base - discount + extra
     * - tax = taxable x taxRate
     * - final = taxable + tax - ADVANCE
     *
     * @param booking The booking for which to calculate the invoice.
     * @return Fully populated E195569Invoice instance.
     */
    // Algorithm: Direct calculation applying hotel billing rules and tiered long-stay discount
    public E195569Invoice calculate(E195569Booking booking) {
        Objects.requireNonNull(booking, "Booking cannot be null.");

        E195569Room room = booking.getRoom();
        E195569RoomType roomType = room.getRoomType();

        double dailyRate = room.getDailyRate();
        long nights = booking.getNights();
        int guests = booking.getNumberOfGuests();
        int capacity = roomType.getCapacity();
        double extraGuestCharge = roomType.getExtraGuestCharge();
        double taxRate = roomType.getTaxRate();
        boolean breakfastIncluded = roomType.isBreakfastIncluded();

        // Business formula implementation
        double base = dailyRate * nights;
        double extra = Math.max(0, guests - capacity) * extraGuestCharge * nights;
        double discount = nights >= DISCOUNT_THRESHOLD_NIGHTS ? DISCOUNT_RATE * base : 0;
        double taxable = base - discount + extra;
        double tax = taxable * taxRate;
        double finalAmount = taxable + tax - ADVANCE;

        return new E195569Invoice(
                booking.getGuest(), room, booking.getCheckInDate(), booking.getCheckOutDate(),
                nights, guests, breakfastIncluded, base, extra, discount, tax, ADVANCE, finalAmount
        );
    }
}

public final class E195569ConsoleReader {

    private static final Scanner SCANNER = new Scanner(System.in);

    /**

     * @param prompt User instruction prompt.
     * @return Trimmed non-empty string entered by the user.
     */
    public static String readString(String prompt) {
        while (true) {
            System.out.print(prompt);
            try {
                if (!SCANNER.hasNextLine()) {
                    return "";
                }
                String input = SCANNER.nextLine();
                if (input != null && !input.trim().isBlank()) {
                    return input.trim();
                }
                System.out.println("  [!] Input cannot be empty. Please try again.");
            } catch (Exception e) {
                System.out.println("  [!] Error reading input. Please try again.");
            }

     * @param max    Maximum allowed value (inclusive).
     * @return Validated integer within the specified range.
     */
    public static int readInt(String prompt, int min, int max) {
        while (true) {
            System.out.print(prompt);
            try {
                if (!SCANNER.hasNextLine()) {
                    return min;

                String input = SCANNER.nextLine();
                if (input == null || input.trim().isBlank()) {

                if (value >= min && value <= max) {

                }
                System.out.printf("  [!] Please enter a number between %d and %d.%n", min, max);
            } catch (NumberFormatException e) {
                System.out.println("  [!] Invalid number format. Please enter a valid integer.");
            } catch (Exception e) {
                System.out.println("  [!] Error reading input. Please try again.");
            }

            }
            System.out.print(displayPrompt);
            try {
                if (!SCANNER.hasNextLine()) {
                    return LocalDate.now();
                }
                String input = SCANNER.nextLine();
                if (input == null || input.trim().isBlank()) {
                    System.out.println("  [!] Date cannot be empty. Please use yyyy-MM-dd (e.g., 2026-12-01).");
                    continue;
                }
                return LocalDate.parse(input.trim(), DATE_FORMATTER);
            } catch (DateTimeParseException e) {
                System.out.println("  [!] Invalid date format. Please use yyyy-MM-dd (e.g., 2026-12-01).");
